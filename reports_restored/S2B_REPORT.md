    # 빈 값: None / '' / 공백만 / [] / list 원소 모두 공백 / {}
    if v is None: return True
    if isinstance(v, str): return v.strip() == ''
    if isinstance(v, list):
        return len(v) == 0 or all(is_empty(x) for x in v)
    if isinstance(v, dict): return len(v) == 0
    return False
